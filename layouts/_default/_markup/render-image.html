{{- $destination := strings.TrimPrefix "./" .Destination -}} {{- $resource := .Page.Resources.GetMatch $destination -}} {{- if and $resource (eq $resource.MediaType.MainType "image") -}} {{- $widths := slice 480 768 1200 -}} {{- $srcset := slice -}} {{- range $width := $widths -}} {{- if lt $width $resource.Width -}} {{- $variant := $resource.Resize (printf "%dx webp q82" $width) -}} {{- $srcset = $srcset | append (printf "%s %dw" $variant.RelPermalink $variant.Width) -}} {{- end -}} {{- end -}} {{- $fallback := $resource.Resize (printf "%dx webp q82" (int $resource.Width)) -}} {{- $srcset = $srcset | append (printf "%s %dw" $fallback.RelPermalink $fallback.Width) -}} {{- else -}} {{- end -}}
